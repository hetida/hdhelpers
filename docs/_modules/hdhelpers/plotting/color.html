42def resolve_color(color_str: str) -> str:
43    """Resolve a Fuseki palette name to a color string.
44
45    Args:
46        color_str: Palette name, e.g. "ki.energy", or any other string.
47
48    Returns:
49        str: Resolved color; other strings are returned unchanged.
50
51    Code example:
52
53    .. doctest::
54
55        >>> resolve_color("red")
56        'red'
57        >>> resolve_color("ki.energy")
58        '#eb6962'
59    """
60    return FUSEKI_COLORS.get(color_str, color_str)
