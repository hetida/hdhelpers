12TICKFORMATSTOPS = [

27def set_dt_ticks(fig: go.Figure) -> None:
28    """Set zoom-dependent, locale-independent date/time labels on all x axes.
29
30    Update tickformatstops; timestamps and time zones are unchanged.
31
32    Args:
33        fig: Figure to format.
34
35    Returns:
36        None: The supplied figure is updated in place.
37
38    Code example:
39
40    .. doctest::
41
42        >>> import plotly.graph_objects as go
43        >>> fig = go.Figure(data=go.Scatter(x=["2024-01-01", "2024-01-02"], y=[1, 4]))
44        >>> set_dt_ticks(fig)
45    """
46    fig.update_xaxes(tickformatstops=TICKFORMATSTOPS)
