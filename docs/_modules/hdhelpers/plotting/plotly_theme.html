48def set_agnostic_theme(fig: go.Figure) -> None:
49    """Apply a Plotly theme suitable for light and dark backgrounds.
50
51    Args:
52        fig: Figure to style.
53
54    Returns:
55        None: The supplied figure is updated in place.
56
57    Code example:
58
59    .. doctest::
60
61        >>> import plotly.graph_objects as go
62        >>> fig = go.Figure(data=go.Scatter(x=[1, 2, 3], y=[1, 4, 9]))
63        >>> set_agnostic_theme(fig)
64    """
65    fig.update_layout(template=theme_agnostic_template)
