Source code for hdhelpers.metadata.attached

  1"""Metadata of the conventions attached to pandas objects as model
  2
  3The runtime of the hetida designer parses the metadata of each workflow input once and attaches
  4the model under attrs["hd_metadata"] (META_KEY), in place of the keys of the conventions
  5("dataset_metadata", "metrics", "value_dimensions_shared"). Other keys of attrs stay as they are.
  6
  7pandas deep-copies attrs for (almost) every object derived from another one. The models are
  8immutable and return themselves when copied, so that all derived objects share the model
  9instead of copying the metadata.
 10
 11Component code reads the metadata via read_metadata and attaches a changed version (see the
 12with_* methods of the models) to its own result via set_metadata. Everything serializing attrs
 13to JSON needs attrs_to_json, since the model is no JSON value.
 14"""

 45def infer_data_type(obj: pd.Series | pd.DataFrame) -> TimeseriesDataType:
 46    """SERIES for a Series, MULTITSFRAME for a frame with a "metric" column, else SINGLETSFRAME"""
 47    if isinstance(obj, pd.Series):
 48        return "SERIES"
 49    return "MULTITSFRAME" if "metric" in obj.columns else "SINGLETSFRAME"

 52def _convention_attrs(attrs: Mapping[Any, Any]) -> dict[str, Any]:

 56def attach_metadata(
 57    obj: pd.Series | pd.DataFrame, data_type: TimeseriesDataType | None = None
 58) -> AnyMetadata | None:
 59    """Parse metadata in attrs and attach the resulting immutable model.
 60
 61    Replace convention fields with attrs["hd_metadata"]; preserve other attrs. An already
 62    attached model is returned unchanged.
 63
 64    Args:
 65        obj: Series or DataFrame to update in place.
 66        data_type:
 67            Convention for raw attrs, or None to infer it; see
 68            :ref:`metadata-type-selection`. Used only when no model is attached.
 69
 70    Returns:
 71        SeriesMetadata | SingleTSFrameMetadata | MultiTSFrameMetadata | None:
 72            Attached model, or None for absent/invalid metadata. Failure leaves attrs
 73            unchanged.
 74
 75    Code example:
 76
 77    Using :ref:`metadata-example-data`, attach a model while preserving unrelated attrs:
 78
 79    .. doctest:: metadata.attach_metadata
 80
 81        >>> data = series.copy()
 82        >>> data.attrs["source"] = "example"
 83        >>> meta = md.attach_metadata(data)
 84        >>> meta.unit()
 85        'm'
 86        >>> sorted(data.attrs)
 87        ['hd_metadata', 'source']
 88        >>> md.read_metadata(data) is meta
 89        True
 90        >>> md.attach_metadata(data) is meta
 91        True
 92        >>> md.attach_metadata(pd.Series([1.0])) is None
 93        True
 94    """
 95    attached = obj.attrs.get(META_KEY)
 96    if isinstance(attached, _TimeseriesMetadataBase):
 97        return attached  # type: ignore[return-value]
 98    convention_attrs = _convention_attrs(obj.attrs)
 99    if len(convention_attrs) == 0:
100        return None
101    model = METADATA_MODELS[data_type or infer_data_type(obj)]
102    try:
103        meta = model.model_validate(convention_attrs)
104    except ValidationError:
105        return None
106    obj.attrs = {key: value for key, value in obj.attrs.items() if key not in CONVENTION_KEYS} | {
107        META_KEY: meta
108    }
109    return meta  # type: ignore[return-value]

144def read_metadata(
145    obj: pd.Series | pd.DataFrame,
146    data_type: TimeseriesDataType | None = None,
147    *,
148    required: bool = False,
149) -> AnyMetadata | None:
150    """Read metadata as a model without changing the input.
151
152    Reuse an attached model; otherwise parse convention fields in attrs.
153
154    Args:
155        obj: Series or DataFrame to read.
156        data_type:
157            Expected convention. None keeps an attached model's type or infers it for raw
158            attrs; see :ref:`metadata-type-selection`.
159        required:
160            If True, raise on missing, invalid or wrong-type metadata instead of returning
161            None.
162
163    Returns:
164        SeriesMetadata | SingleTSFrameMetadata | MultiTSFrameMetadata | None:
165            Attached or newly parsed model; None for missing, invalid or wrong-type
166            metadata.
167
168    Raises:
169        MetadataConventionError: Metadata cannot be returned and required=True.
170
171    Code example:
172
173    Using :ref:`metadata-example-data`, read a model without changing the original attrs:
174
175    .. doctest:: metadata.read_metadata
176
177        >>> meta = md.read_metadata(series, required=True)
178        >>> meta.unit()
179        'm'
180        >>> "hd_metadata" in series.attrs
181        False
182        >>> missing = pd.Series([1.0])
183        >>> md.read_metadata(missing) is None
184        True
185        >>> try:
186        ...     md.read_metadata(missing, required=True)
187        ... except md.MetadataConventionError as error:
188        ...     print(error.issues[0].message)
189        No metadata in attrs.
190    """
191    attached = obj.attrs.get(META_KEY)
192    if isinstance(attached, _TimeseriesMetadataBase):
193        if data_type is None or attached._data_type == data_type:
194            return attached  # type: ignore[return-value]
195        _no_metadata(
196            required,
197            f"The attached metadata is {attached._data_type} metadata, not {data_type} metadata.",
198        )
199        return None
200
201    convention_attrs = _convention_attrs(obj.attrs)
202    if len(convention_attrs) == 0:
203        _no_metadata(required, "No metadata in attrs.")
204        return None
205    data_type = data_type or infer_data_type(obj)
206    try:
207        return METADATA_MODELS[data_type].model_validate(convention_attrs)  # type: ignore[return-value]
208    except ValidationError as e:
209        if not required:
210            return None
211        issues = [
212            issue
213            for issue in check_metadata_conventions(convention_attrs, data_type)
214            if issue.severity == MetadataIssueSeverity.ERROR
215        ]
216        raise MetadataConventionError(issues) from e

219def _no_metadata(required: bool, message: str) -> None:
220    """None, or with required raise a MetadataConventionError"""
221    if required:
222        raise MetadataConventionError(
223            [
224                MetadataConventionIssue(
225                    severity=MetadataIssueSeverity.ERROR, location="", message=message
226                )
227            ]
228        )

231def set_metadata(obj: pd.Series | pd.DataFrame, meta: AnyMetadata) -> None:
232    """Attach the supplied metadata model to a result object.
233
234    Store it under attrs["hd_metadata"], replacing convention fields and preserving
235    unrelated attrs.
236
237    Args:
238        obj: Result Series or DataFrame created by your component; updated in place.
239        meta: Validated metadata model; shared without copying or revalidation.
240
241    Returns:
242        None: Updates obj in place.
243
244    Code example:
245
246    Convert the metre values from :ref:`metadata-example-data` to centimetres and set matching
247    metadata on the result. The original metadata stays unchanged:
248
249    .. doctest:: metadata.set_metadata
250
251        >>> meta = md.read_metadata(series, required=True)
252        >>> result = series * 100
253        >>> changed = meta.with_value_dimension(None, "value", unit="cm")
254        >>> md.set_metadata(result, changed)
255        >>> md.get_series_unit(result), md.get_series_unit(series)
256        ('cm', 'm')
257        >>> md.read_metadata(result) is changed
258        True
259    """
260    obj.attrs = {key: value for key, value in obj.attrs.items() if key not in CONVENTION_KEYS} | {
261        META_KEY: meta
262    }

265def attrs_to_json(attrs: Mapping[Any, Any]) -> dict[Any, Any]:
266    """Replace attached metadata with its JSON fields in a new attrs dictionary.
267
268    Keep unrelated attrs as-is. Omit optional metadata fields set to None or their defaults.
269
270    Args:
271        attrs: Pandas attrs mapping; left unchanged.
272
273    Returns:
274        dict[Any, Any]:
275            Dictionary for json.dumps or export. Unrelated attrs must also be
276            JSON-compatible.
277
278    Code example:
279
280    Using :ref:`metadata-example-data`, convert attached models to JSON-compatible dictionaries.
281    This function returns a dictionary; json.dumps produces the JSON string:
282
283    .. doctest:: metadata.attrs_to_json
284
285        >>> import json
286        >>> data = series.copy()
287        >>> _ = md.attach_metadata(data)
288        >>> attrs = md.attrs_to_json(data.attrs)
289        >>> isinstance(attrs, dict), "hd_metadata" in attrs
290        (True, False)
291        >>> encoded = json.dumps(attrs)
292        >>> json.loads(encoded)["metrics"][0]["value_dimensions"][0]["unit"]
293        'm'
294        >>> export = data.copy()
295        >>> export.attrs = md.attrs_to_json(export.attrs)
296        >>> "hd_metadata" in data.attrs, "hd_metadata" in export.attrs
297        (True, False)
298
299    The export copy can now be written with ``export.to_frame().to_parquet("series.parquet")``
300    if a pandas Parquet engine is installed. The original keeps its attached model.
301    """
302    attached = attrs.get(META_KEY)
303    if not isinstance(attached, _TimeseriesMetadataBase):
304        return dict(attrs)
305    rest = {key: value for key, value in attrs.items() if key != META_KEY}
306    return rest | attached.model_dump(mode="json")
