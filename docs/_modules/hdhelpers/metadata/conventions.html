   1"""Metadata conventions for timeseries data
   2
   3The models in this module specify the hetida designer metadata conventions for the metadata
   4(attrs) of SERIES, SINGLETSFRAME and MULTITSFRAME objects, one model per data type. See the user
   5guide page on attached metadata for an introduction.
   6
   7Raw metadata is stored as convention fields in pandas attrs. A parsed model can be attached
   8under attrs["hd_metadata"]; read_metadata accepts both forms. Adapters exchange the raw form,
   9e.g. in the Data-Attributes header of generic REST adapter timeseries endpoints or the
  10__metadata__ field of the direct provisioning adapter's wrapped format. Use attrs_to_json
  11to convert attached models back to raw fields for export.
  12
  13The conventions are strict: Metadata adhering to them provides the required fields, possibly as
  14null if the information is unknown, e.g. the unit of a value dimension. Consumers like components
  15should nevertheless cope with missing metadata, since validation of metadata received from
  16adapters is optional.
  17
  18The models document the request body of the hetida designer backend endpoint for checking
  19metadata. Hence they occur as JSON Schemas in the openapi.json of the backend, so that third party
  20projects, e.g. adapters providing or receiving such metadata, can validate against them.
  21Additional fields are allowed at every level. The models are immutable, see ConventionModel.
  22Metadata timestamps are UTC ISO strings with an explicit Z or +00:00 offset.
  23
  24check_metadata_conventions additionally checks the rules which JSON Schema cannot express,
  25optionally against the actual data.
  26"""

  87def _check_point_in_time(timestamp: str) -> str:
  88    """Reject timestamps of the pattern which are no actual points in time, e.g. February 30"""
  89    try:
  90        datetime.datetime.fromisoformat(timestamp)
  91    except ValueError as e:
  92        raise ValueError(f"{timestamp!r} is not an actual point in time: {e}.") from e
  93    return timestamp

  96def _utc_isoformat(value: Any) -> Any:
  97    """A datetime with UTC offset 0 as isoformat string, so that producers can pass datetimes"""
  98    if not isinstance(value, datetime.datetime):
  99        return value
 100    if value.utcoffset() is None:
 101        raise ValueError(f"{value!r} is naive, but needs to be in UTC.")
 102    if value.utcoffset() != datetime.timedelta(0):
 103        raise ValueError(f"{value!r} is not in UTC.")
 104    return value.isoformat()

 107# The pattern specifies the format in the JSON Schema, which cannot express that the date and

 199
 200    Holds dict values of extra fields of the convention models, since the models are shared

 219def _freeze(value: Any) -> Any:
 220    """A JSON value with dicts and lists as FrozenDicts and tuples, recursively"""
 221    if isinstance(value, dict):
 222        if not all(isinstance(key, str) for key in value):
 223            raise ValueError("Keys of JSON objects must be strings.")
 224        return FrozenDict({key: _freeze(item) for key, item in value.items()})
 225    if isinstance(value, list | tuple):
 226        return tuple(_freeze(item) for item in value)
 227    if (
 228        value is None
 229        or isinstance(value, str | int)
 230        or (isinstance(value, float) and math.isfinite(value))
 231    ):
 232        return value
 233    raise ValueError(f"{value!r} is not a JSON value.")

 255def _item_count(min_length: int, max_length: int | None = None) -> BeforeValidator:
 256    """Check the number of items before validating them
 257
 258    Pydantic counts the items of a tuple after validating them, so that an invalid item would
 259    also be reported as missing. The JSON Schema states the counts via json_schema_extra.
 260    """
 261
 262    def items(count: int) -> str:
 263        return f"{count} item" if count == 1 else f"{count} items"
 264
 265    def check(value: Any) -> Any:
 266        if isinstance(value, list | tuple):
 267            if len(value) < min_length:
 268                raise PydanticCustomError(
 269                    "too_short", f"Should have at least {items(min_length)}, not {len(value)}"
 270                )
 271            if max_length is not None and len(value) > max_length:
 272                raise PydanticCustomError(
 273                    "too_long", f"Should have at most {items(max_length)}, not {len(value)}"
 274                )
 275            # a tuple, since strict validation after a before validator rejects lists
 276            return tuple(value)
 277        return value
 278
 279    return BeforeValidator(check)

 282class ConventionModel(BaseModel):
 283    """Base of the models of the metadata conventions
 284
 285    The models are immutable, including the values of extra fields, so that pandas objects can
 286    share them instead of copying them: pandas deep-copies the attrs of an object for (almost)
 287    every object derived from it. A changed version of a model is a new model, which shares all
 288    unchanged parts with the original.
 289    """
 290
 291    # Validation is lax, so that Python input may contain lists for the tuples. Fields whose type
 292    # lax validation would coerce (bool, int, float) use the strict types instead, so that the
 293    # results equal strict validation of JSON input.
 294    model_config = ConfigDict(extra="allow", frozen=True)
 295    __pydantic_extra__: dict[str, _FrozenValue]
 296
 297    def __deepcopy__(self, memo: dict[int, Any] | None = None) -> Self:
 298        return self
 299
 300    def model_copy(
 301        self,
 302        *,
 303        update: Mapping[str, Any] | None = None,
 304        deep: bool = False,  # noqa: ARG002
 305    ) -> Self:
 306        """Copy with the given fields replaced, without validating them
 307
 308        Always shallow, since the model is immutable. A deep copy would apply the update to the
 309        original, since __deepcopy__ returns the original.
 310        """
 311        return super().model_copy(update=update, deep=False)
 312
 313    @classmethod
 314    def __get_pydantic_json_schema__(
 315        cls, core_schema: CoreSchema, handler: GetJsonSchemaHandler
 316    ) -> JsonSchemaValue:
 317        json_schema = handler(core_schema)
 318        # Any JSON value is allowed in extra fields, as without the typed __pydantic_extra__
 319        handler.resolve_ref_schema(json_schema)["additionalProperties"] = True
 320        return json_schema

 700class _TimeseriesMetadataBase(MetadataReading, ConventionModel):
 701    """Base of the root models, the metadata of a SERIES, SINGLETSFRAME or MULTITSFRAME
 702
 703    The methods of MetadataReading look up information with the fallbacks of the conventions.
 704    The with_* methods return changed versions, which share all unchanged parts with this one.
 705    """
 706
 707    _data_type: ClassVar[TimeseriesDataType]
 708
 709    value_dimensions_shared: tuple[SharedValueDimensionMetadata, ...] | None = optional(
 710        description="Metadata of value dimensions that have the same meaning for all metrics."
 711        " Consumers use it as fallback, but it does not replace the value dimensions of the"
 712        " metrics. Each column may occur at most once.",
 713    )

 715    @model_validator(mode="wrap")
 716    @classmethod
 717    def _conforms(
 718        cls, data: Any, handler: ModelWrapValidatorHandler[Self], info: ValidationInfo
 719    ) -> Self:
 720        """Reject metadata violating the rules of the conventions which JSON Schema cannot express
 721
 722        check_metadata_conventions reports these errors itself, at their exact locations.
 723        """
 724        model = handler(data)
 725        if (info.context or {}).get(_SKIP_RULES) is True:
 726            return model
 727        errors = _rule_errors(data, model, cls._data_type)
 728        if len(errors) > 0:
 729            raise MetadataConventionError(errors)
 730        return model

 733        copied = super().model_copy(update=update, deep=deep)
 734        copied.__dict__.pop("_metrics_by_id", None)  # the metrics may differ
 735        return copied

 738        """The metadata as model with the given fields, keeping the extra fields"""
 739        return model.model_validate(self.__pydantic_extra__ | fields)

 741    def with_metric(self, metric_id: str | None = None, /, **changes: Any) -> Self:
 742        """A version with fields of a metric changed, the other metrics are shared
 743
 744        The changed metric is validated, including the rules of the conventions which concern it.
 745        """
 746        old = self.metric(metric_id)
 747        new = type(old).model_validate(old.model_dump() | changes)
 748        errors = _metric_rule_errors(self, new)
 749        new_id = id_of(new, self.dataset_metadata.metric_key)
 750        if new_id != id_of(old, self.dataset_metadata.metric_key) and new_id in self.metric_ids():

 752        if len(errors) > 0:

 755            update={"metrics": tuple(new if metric is old else metric for metric in self.metrics)}
 756        )

1151            # e.g. "nan" gives NaT

1154            return False

1158@lru_cache(maxsize=1024)
1159def _duration(value: str) -> pd.Timedelta | None:
1160    """The fixed duration of a frequency, None if calendar based like "MS" or no frequency"""
1161    try:
1162        offset = pd.tseries.frequencies.to_offset(value)
1163    except ValueError:
1164        try:
1165            # e.g. "P1D" or "1 days"
1166            duration = pd.to_timedelta(value)
1167        except ValueError:
1168            return None
1169        return None if pd.isna(duration) else duration
1170    if isinstance(offset, pd.offsets.Day):
1171        # a calendar day since pandas 3, i.e. 24 hours in UTC
1172        return pd.Timedelta(days=offset.n)
1173    if isinstance(offset, pd.offsets.Tick):
1174        return pd.Timedelta(offset)
1175    # calendar based, e.g. months
1176    return None

1192def _check_frequency(
1193    obj: dict[str, Any], field: str, location: str
1194) -> list[MetadataConventionIssue]:
1195    """Check that a frequency field is a positive frequency"""
1196    value = obj.get(field)
1197    if value is None:
1198        return []

1200        return [_not_a_frequency(f"{location}.{field}", value)]
1201    duration = _duration(value)
1202    positive = duration > pd.Timedelta(0) if duration is not None else _offset_count(value) > 0
1203    return [] if positive else [_error(f"{location}.{field}", f'"{value}" is not positive.')]

1206def _check_offset(
1207    obj: dict[str, Any], frequency_field: str, offset_field: str, location: str
1208) -> list[MetadataConventionIssue]:
1209    """Check the offset of the timestamps against the intervals of a valid frequency"""
1210    offset = obj[offset_field]
1211    offset_location = f"{location}.{offset_field}"
1212    if not _is_frequency(offset):
1213        return [_not_a_frequency(offset_location, offset)]
1214    offset_duration = _duration(offset)
1215    if offset_duration is None or offset_duration < pd.Timedelta(0):
1216        return [
1217            _error(offset_location, f'"{offset}" is not a non-negative timedelta, e.g. "4min".')
1218        ]
1219    # None for calendar based frequencies, which offsets cannot be compared with
1220    frequency_duration = _duration(obj[frequency_field])
1221    if frequency_duration is not None and offset_duration >= frequency_duration:
1222        return [
1223            _error(
1224                offset_location,
1225                f'"{offset}" is not smaller than {frequency_field} "{obj[frequency_field]}".',
1226            )
1227        ]
1228    return []

1231def _check_frequency_with_alignment(
1232    obj: dict[str, Any], frequency_field: str, offset_field: str, timezone_field: str, location: str
1233) -> list[MetadataConventionIssue]:
1234    """Check a frequency and the fields aligning its intervals, the offset and the time zone
1235
1236    They are only meaningful together with the frequency on the same level, since the frequency
1237    of a metric falls back to the one of the dataset metadata as a whole.
1238    """
1239    issues = _check_frequency(obj, frequency_field, location)
1240    valid_frequency = len(issues) == 0 and obj.get(frequency_field) is not None
1241    for field in (offset_field, timezone_field):
1242        if obj.get(field) is not None and obj.get(frequency_field) is None:
1243            issues.append(_error(f"{location}.{field}", f"Requires {frequency_field}."))
1244    issues += _check_timezone(obj, location, timezone_field)
1245    if obj.get(offset_field) is not None and valid_frequency:
1246        issues += _check_offset(obj, frequency_field, offset_field, location)
1247    return issues

1337def _unknown_field_issues(
1338    obj: dict[str, Any], model: type[BaseModel], location: str, ignored: set[str] | None = None
1339) -> list[MetadataConventionIssue]:
1340    """Warnings for fields not defined by the conventions, which may be misspelled"""
1341    known = _known_fields(model)
1342    issues = []
1343    for field in obj:
1344        if field in known or (ignored is not None and field in ignored):
1345            continue
1346        hint = _MISPLACED_FIELD_HINTS.get((model, field))
1347        if hint is None:
1348            snake_case = re.sub(r"(?<!^)(?=[A-Z])", "_", field).lower()
1349            close_matches = (
1350                [snake_case]
1351                if snake_case in known
1352                else difflib.get_close_matches(field, known, n=1, cutoff=0.8)
1353            )
1354            hint = (
1355                f'Did you mean "{close_matches[0]}"?'
1356                if len(close_matches) > 0
1357                else "Additional fields are allowed, but make sure it is not misspelled."
1358            )
1359        issues.append(
1360            _warning(
1361                f"{location}.{field}" if location != "" else field,
1362                f'Unknown field "{field}". {hint}',
1363            )
1364        )
1365    return issues

1432def _check_consistency(attrs: dict[str, Any]) -> list[MetadataConventionIssue]:
1433    """Check the rules of the conventions which the JSON Schema cannot express
1434
1435    Expects attrs to be valid against the JSON Schema of the data type.
1436    """
1437    issues: list[MetadataConventionIssue] = []
1438    dataset_metadata = attrs["dataset_metadata"]
1439    metrics = attrs["metrics"]
1440    metric_key = dataset_metadata["metric_key"]
1441
1442    metric_ids = []
1443    for index, metric in enumerate(metrics):
1444        metric_id = metric.get(metric_key)
1445        if not isinstance(metric_id, str):
1446            issues.append(
1447                _error(
1448                    f"metrics[{index}]",
1449                    f'Metric has no string field "{metric_key}", which identifies metrics'
1450                    " according to dataset_metadata.metric_key.",
1451                )
1452            )
1453        else:
1454            metric_ids.append(metric_id)
1455        issues += [
1456            _warning(f"metrics[{index}].{name}", f"Deprecated. {deprecated}")
1457            for name, deprecated in _DEPRECATED_METRIC_FIELDS.items()
1458            if metric.get(name) is not None
1459        ]
1460        issues += _check_duplicate_columns(
1461            metric["value_dimensions"], f"metrics[{index}].value_dimensions"
1462        )
1463        issues += _check_frequency_with_alignment(
1464            metric,
1465            "data_frequency",
1466            "data_frequency_offset",
1467            "data_frequency_timezone",
1468            f"metrics[{index}]",
1469        )
1470        issues += _check_timezone(metric, f"metrics[{index}]")
1471        issues += _check_recording(metric, dataset_metadata, f"metrics[{index}]")
1472        for comment_index, comment in enumerate(metric.get("comments") or []):
1473            issues += _check_interval(
1474                comment, f"metrics[{index}].comments[{comment_index}]", "start_date", "end_date"
1475            )
1476    issues += [
1477        _error("metrics", f'Metric "{metric_id}" occurs more than once.')
1478        for metric_id, count in Counter(metric_ids).items()
1479        if count > 1
1480    ]
1481    issues += _check_duplicate_columns(
1482        attrs.get("value_dimensions_shared") or [], "value_dimensions_shared"
1483    )
1484
1485    single_metric = dataset_metadata.get("single_metric")
1486    if single_metric is not None and len(metrics) > 1:
1487        issues.append(
1488            _error(
1489                "dataset_metadata.single_metric",
1490                f"Must not be set, since {len(metrics)} metrics are present under metrics.",
1491            )
1492        )
1493    elif single_metric is not None and len(metric_ids) == 1 and single_metric not in metric_ids:
1494        issues.append(
1495            _error(
1496                "dataset_metadata.single_metric",
1497                f'"{single_metric}" is not the "{metric_key}" of the metric under metrics.',
1498            )
1499        )
1500    elif single_metric is not None and single_metric not in dataset_metadata["queried_metrics"]:
1501        issues.append(_error("dataset_metadata.single_metric", "Missing in queried_metrics."))
1502
1503    if dataset_metadata.get("only_invalidate") is True and (
1504        dataset_metadata.get("invalidate_dataset") is False
1505    ):
1506        issues.append(
1507            _error(
1508                "dataset_metadata.only_invalidate",
1509                "Cannot be true if invalidate_dataset is false.",
1510            )
1511        )
1512
1513    if dataset_metadata.get("ref_dataset_discrete") is not True:
1514        issues += [
1515            _error(
1516                f"dataset_metadata.{field}",
1517                "Required unless dataset_metadata.ref_dataset_discrete is true.",
1518            )
1519            for field in REF_INTERVAL_FIELDS
1520            if dataset_metadata.get(field) is None
1521        ]
1522    issues += _check_interval(
1523        dataset_metadata,
1524        "dataset_metadata",
1525        "ref_interval_start_timestamp",
1526        "ref_interval_end_timestamp",
1527    )
1528    issues += _check_interval(
1529        dataset_metadata,
1530        "dataset_metadata",
1531        "invalidation_interval_start",
1532        "invalidation_interval_end",
1533    )
1534    issues += _check_frequency_with_alignment(
1535        dataset_metadata,
1536        "ref_data_frequency",
1537        "ref_data_frequency_offset",
1538        "ref_data_frequency_timezone",
1539        "dataset_metadata",
1540    )
1541    issues += _check_frequency(dataset_metadata, "default_aggregation_interval", "dataset_metadata")
1542    issues += _check_timezone(dataset_metadata, "dataset_metadata")
1543    return issues

1556def _check_data_in_ref_interval(
1557    dataset_metadata: dict[str, Any], data: pd.Series | pd.DataFrame
1558) -> list[MetadataConventionIssue]:
1559    """Warn if the data has timestamps outside of the ref interval"""
1560    start = dataset_metadata.get("ref_interval_start_timestamp")
1561    end = dataset_metadata.get("ref_interval_end_timestamp")
1562    interval_type = dataset_metadata.get("ref_interval_type")
1563    timestamps = data.index if isinstance(data, pd.Series) else data.get("timestamp")
1564    if (
1565        dataset_metadata.get("ref_dataset_discrete") is True
1566        or start is None
1567        or end is None
1568        or interval_type is None
1569        or pd.Timestamp(start) > pd.Timestamp(end)
1570        or timestamps is None
1571        or len(timestamps) == 0
1572        or not pd.api.types.is_datetime64_any_dtype(timestamps)
1573    ):
1574        return []
1575    ref_interval = pd.Interval(
1576        pd.Timestamp(start), pd.Timestamp(end), closed=_CLOSED_BY_INTERVAL_TYPE[interval_type]
1577    )
1578    # naive timestamps are considered to be UTC
1579    first, last = (
1580        timestamp if timestamp.tzinfo is not None else timestamp.tz_localize("UTC")
1581        for timestamp in (timestamps.min(), timestamps.max())
1582    )
1583    if first in ref_interval and last in ref_interval:
1584        return []
1585    return [
1586        _warning(
1587            "dataset_metadata",
1588            f"The data has timestamps outside of the {interval_type} ref interval: The first is"
1589            f" {first.isoformat()}, the last {last.isoformat()}.",
1590        )
1591    ]

1622                    f" {_listing(unqueried)}",

1626    undescribed_metrics = sorted(
1627        set(dataset_metadata["queried_metrics"]) - metric_ids - set(data_metrics)

1631            _warning(

1739
1740    For pandas inputs, also check metric IDs, value columns and time bounds. An empty list

1742
1743    Args:
1744        obj: Series/DataFrame to check, or a mapping containing metadata.
1745        data_type:
1746            Convention to check; required for mappings. None infers it from pandas input;
1747            see :ref:`metadata-type-selection`.
1748
1749    Returns:
1750        list[MetadataConventionIssue]:
1751            Issues with severity, location and message. Invalid metadata is reported as
1752            issues.
1753
1754    Raises:
1755        ValueError: A metadata mapping is provided without data_type.
1756
1757    Code example:
1758
1759    Using :ref:`metadata-example-data`, validate the data and its metadata together, or pass
1760    plain metadata with an explicit data type. Issues describe the severity, location and cause:
1761
1762    .. doctest:: metadata.check_metadata_conventions
1763
1764        >>> md.check_metadata_conventions(series)
1765        []
1766        >>> md.check_metadata_conventions(series.attrs, "SERIES")
1767        []
1768        >>> invalid = series.copy()
1769        >>> invalid.attrs["metrics"][0]["default_timezone"] = "Mars/Base"
1770        >>> issues = md.check_metadata_conventions(invalid)
1771        >>> [(issue.severity.value, issue.location) for issue in issues]
1772        [('error', 'metrics[0].default_timezone')]
1773        >>> print(issues[0].message)  # doctest: +ELLIPSIS
1774        "Mars/Base" is neither an IANA time zone name like "Europe/Berlin" nor a UTC offset...
1775    """
1776    # imported here, since hdhelpers.metadata.attached imports this module
1777    from hdhelpers.metadata.attached import attrs_to_json  # noqa: PLC0415
1778
1779    data = obj if isinstance(obj, pd.Series | pd.DataFrame) else None
1780    attrs = data.attrs if data is not None else obj
1781    if isinstance(attrs, Mapping):
1782        # attached metadata in its JSON form
1783        attrs = attrs_to_json(attrs)
1784    if data_type is None and isinstance(data, pd.Series):
1785        data_type = "SERIES"
1786    elif data_type is None and isinstance(data, pd.DataFrame):
1787        data_type = "MULTITSFRAME" if "metric" in data.columns else "SINGLETSFRAME"
1788    elif data_type is None:
1789        raise ValueError("The data type is required for checking metadata without data.")
1790    model = METADATA_MODELS[data_type]
1791
1792    try:
1793        attrs_json = json.dumps(attrs, allow_nan=False)
1794    except (TypeError, ValueError) as e:
1795        return [
1796            _error(
1797                "",
1798                f"Metadata is not JSON serializable, so it cannot be exchanged with adapters: {e}",
1799            )
1800        ]
1801

1803        model.model_validate_json(attrs_json, context={_SKIP_RULES: True})
1804    except ValidationError as e:

1810                    "Should be a UTC isoformat timestamp with explicit offset, i.e. ending with Z"
1811                    f" or +00:00, got {error['input']!r}."
1812                )
1813            elif (
1814                error["loc"] == ("metrics",)
1815                and isinstance(attrs, Mapping)
1816                and isinstance(attrs.get("metrics"), dict)
1817            ):
1818                message += (
1819                    ". Metrics as a mapping is an older metadata convention, the current one"
1820                    " expects a list of metric objects."
1821                )
