Source code for hdhelpers.metadata.helpers

  1"""Functions accessing the metadata of SERIES, SINGLETSFRAMEs and MULTITSFRAMEs
  2
  3The functions are thin wrappers over the methods of the models of the metadata conventions,
  4see read_metadata. The methods resolve the fallbacks of the conventions, e.g. a missing display
  5name falls back to the name, see hdhelpers.metadata._reading.
  6
  7The metadata is the model attached to the object (see attach_metadata), whatever data type it
  8belongs to, e.g. the metadata of a SINGLETSFRAME on a Series taken from it. Otherwise the
  9metadata in attrs is parsed as the data type of the function. Metadata which does not follow
 10the conventions gives no information: None or empty defaultdicts. For a SERIES or SINGLETSFRAME,
 11the single metric is the one named by dataset_metadata.single_metric, otherwise the only one.
 12
 13Fields of the functions taking one can be dotted paths, e.g. "inherited.site".
 14"""

 51def _metadata(data: pd.Series | pd.DataFrame, data_type: TimeseriesDataType) -> AnyMetadata | None:
 52    """The attached metadata of any data type, otherwise the attrs parsed as data_type"""
 53    attached = data.attrs.get(META_KEY)
 54    if isinstance(attached, _TimeseriesMetadataBase):
 55        return attached  # type: ignore[return-value]
 56    return read_metadata(data, data_type)

 59def _single_metric(meta: AnyMetadata | None) -> str | None:
 60    """The single metric: dataset_metadata.single_metric if described, otherwise the only one"""
 61    if meta is None:
 62        return None
 63    metric_ids = meta.metric_ids()
 64    single_metric = meta.dataset_metadata.single_metric
 65    if single_metric is not None:
 66        return single_metric if single_metric in metric_ids else None
 67    return metric_ids[0] if len(metric_ids) == 1 else None

 70def _by_metric_by_column(
 71    data: pd.DataFrame, fields: str | tuple[str, ...], include_inherited: bool
 72) -> defaultdict[str, defaultdict[str, Any]]:
 73    check_dataframe(data)
 74    meta = _metadata(data, "MULTITSFRAME")
 75    infos = (
 76        {}
 77        if meta is None
 78        else meta.value_dimension_infos(fields, include_inherited=include_inherited)
 79    )
 80    return defaultdict(
 81        lambda: defaultdict(lambda: None),
 82        {metric: defaultdict(lambda: None, info) for metric, info in infos.items()},
 83    )

 86def _single_metric_by_column(
 87    data: pd.DataFrame, fields: str | tuple[str, ...], include_inherited: bool
 88) -> defaultdict[str, Any]:
 89    check_dataframe(data)
 90    meta = _metadata(data, "SINGLETSFRAME")
 91    metric = _single_metric(meta)
 92    if meta is None or metric is None:
 93        return defaultdict(lambda: None)
 94    return defaultdict(
 95        lambda: None,
 96        meta.value_dimension_infos(fields, include_inherited=include_inherited)[metric],
 97    )

100def _series_value(series: pd.Series, fields: str | tuple[str, ...], include_inherited: bool) -> Any:
101    check_series(series)
102    meta = _metadata(series, "SERIES")
103    metric = _single_metric(meta)
104    if meta is None or metric is None:
105        return None
106    return meta.value_dimension_info(fields, metric, "value", include_inherited=include_inherited)

109def _single_metric_or_dataset(
110    data: pd.Series | pd.DataFrame,
111    data_type: TimeseriesDataType,
112    of_metric: Callable[[AnyMetadata, str], Any],
113    of_dataset: Callable[[AnyMetadata], Any],
114    empty: Any,
115) -> Any:
116    """Information of the single metric, of the dataset metadata if the metric is not found"""
117    meta = _metadata(data, data_type)
118    if meta is None:
119        return empty
120    metric = _single_metric(meta)
121    return of_dataset(meta) if metric is None else of_metric(meta, metric)

246def get_metric_info(
247    multitsframe: pd.DataFrame, metric_info: str, *, include_inherited: bool = True
248) -> defaultdict[str, Any]:
249    """Read a metadata field for each metric.
250
251    Search the metric, its "value" column, shared "value" metadata, then inherited fields
252    (if enabled).
253
254    Args:
255        multitsframe: DataFrame with per-metric metadata in attrs.
256        metric_info: Field name or dotted path, e.g. "sensor.model".
257        include_inherited: Use inherited fields as a last fallback.
258
259    Returns:
260        defaultdict[str, Any]: Field values by metric ID; unresolved fields are None.
261
262    Code example:
263
264    .. doctest:: metadata.get_metric_info
265
266        >>> from hdhelpers.metadata.conventions import MULTITSFRAME_METADATA_EXAMPLE
267        >>> multitsframe.attrs = MULTITSFRAME_METADATA_EXAMPLE
268        >>> md.get_metric_info(multitsframe, "recording_mode")["plant_a.temperature"]
269        'on_change'
270
271    See :ref:`metadata-getter-fields` for shared examples and return shapes.
272    """
273    check_dataframe(multitsframe)
274    meta = _metadata(multitsframe, "MULTITSFRAME")
275    infos = (
276        {} if meta is None else meta.metric_infos(metric_info, include_inherited=include_inherited)
277    )
278    return defaultdict(lambda: None, infos)

566    Args:

568        metric_info: Field name or dotted path, e.g. "sensor.model".

570

572        Any: One field value, or None when unavailable.

601def get_queried_interval(
602    data: pd.Series | pd.DataFrame,
603) -> tuple[datetime.datetime | None, datetime.datetime | None]:
604    """Read the requested UTC time interval from dataset metadata.
605
606    Args:
607        data: Series or DataFrame with dataset metadata in attrs.
608
609    Returns:
610        tuple[datetime.datetime | None, datetime.datetime | None]:
611            (start, end) as timezone-aware datetimes, or (None, None) if unavailable.
612
613    Code example:
614
615    .. doctest:: metadata.get_queried_interval
616
617        >>> from hdhelpers.metadata.conventions import SERIES_METADATA_EXAMPLE
618        >>> series.attrs = SERIES_METADATA_EXAMPLE
619        >>> start, end = md.get_queried_interval(series)
620        >>> start.isoformat()
621        '2026-01-01T00:00:00+00:00'
622
623    See :ref:`metadata-getter-values` for shared examples and return shapes.
624    """
625    check_series_or_dataframe(data)
626    meta = _metadata(data, infer_data_type(data))
627    return (None, None) if meta is None else meta.queried_interval()

630def get_default_timezones(multitsframe: pd.DataFrame) -> defaultdict[str, str | None]:
631    """Read the time zone recommended for calculations on each metric.
632
633    Use the metric's default_timezone, then the dataset's default_timezone.
634
635    Args:
636        multitsframe: DataFrame with per-metric metadata in attrs.
637
638    Returns:
639        defaultdict[str, str | None]:
640            Zone names or UTC offsets by metric ID, or None for unresolved zones. Missing
641            IDs use the dataset default.
642
643    Code example:
644
645    .. doctest:: metadata.get_default_timezones
646
647        >>> from hdhelpers.metadata.conventions import MULTITSFRAME_METADATA_EXAMPLE
648        >>> multitsframe.attrs = MULTITSFRAME_METADATA_EXAMPLE
649        >>> md.get_default_timezones(multitsframe)["plant_a.pressure"]
650        'Europe/Berlin'
651
652    See :ref:`metadata-getter-timezones` for shared examples and return shapes.
653    """

866def copy_without_metadata[T: (pd.Series, pd.DataFrame)](data: T) -> T:
867    """Copy a Series or DataFrame with all attrs removed.
868
869    Data is shared until either object is modified (pandas copy on write).
870
871    Args:
872        data (pd.Series | pd.DataFrame): Source object; left unchanged.
873
874    Returns:
875        pd.Series | pd.DataFrame: Copy with empty attrs.
876
877    Code example:
878
879    .. doctest:: metadata.copy_without_metadata
880
881        >>> multitsframe.attrs = {"dataset_metadata": {"metric_key": "id"}, "metrics": []}
882        >>> data = md.copy_without_metadata(multitsframe)
883        >>> data.attrs
884        {}
885        >>> multitsframe.attrs
886        {'dataset_metadata': {'metric_key': 'id'}, 'metrics': []}
887    """
888    check_series_or_dataframe(data)
889    # The constructor does not copy attrs, in contrast to e.g. copy(deep=False)
890    copy = type(data)(data)
891    copy.attrs = {}
892    return copy
