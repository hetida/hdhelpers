 24@_convert_to_optional_timezone.register(pd.Timestamp | pd.DatetimeIndex | datetime.datetime)
 25def _[T: (pd.Timestamp, pd.DatetimeIndex)](object_to_convert: T, to_timezone: str | None) -> T:
 26    object_to_convert = pd.to_datetime(object_to_convert)
 27    if to_timezone is None:
 28        if object_to_convert.tz is None:
 29            return object_to_convert.tz_localize("UTC")
 30        return object_to_convert
 31    if object_to_convert.tz is None:
 32        return object_to_convert.tz_localize(to_timezone)
 33    return object_to_convert.tz_convert(to_timezone)

 47def modify_timezone[T: (pd.Timestamp, datetime.datetime, pd.Series, pd.DataFrame)](  # noqa: PLR0912
 48    object_to_convert: T,
 49    to_timezone: str | None = None,
 50    column_names: list[str] | None = None,
 51    convert_index: bool = True,
 52) -> T:
 53    """Convert aware timestamps or localize naive timestamps to the selected time zone.
 54
 55    Args:
 56        object_to_convert (pd.Timestamp | datetime.datetime | pd.Series | pd.DataFrame):
 57            Timestamp, datetime, Series or DataFrame to convert.
 58        to_timezone:
 59            Target zone, e.g. "Europe/Berlin". None uses the plot target setting; if unset,
 60            naive values use UTC and aware values keep their zone.
 61        column_names:
 62            Columns to convert. None/[] selects the Series index or DataFrame "timestamp"
 63            column. For Series values, pass [series.name]. Ignored for scalars.
 64        convert_index:
 65            Also convert the pandas index. With column_names=None/[], a Series index is
 66            always converted. Ignored for scalars.
 67
 68    Returns:
 69        pd.Timestamp | pd.Series | pd.DataFrame:
 70            Timestamp for scalar inputs; converted copy for nonempty pandas inputs. Empty
 71            pandas inputs are returned unchanged.
 72
 73    Raises:
 74        TypeError: Unsupported input or incompatible timestamp/index values.
 75        ValueError: Unknown time zone or invalid datetime values.
 76        KeyError: A requested column is absent.
 77
 78    Code example:
 79
 80    .. doctest::
 81
 82        >>> from hdhelpers.helpers import modify_timezone
 83        >>> modified_timezone = modify_timezone(pd.to_datetime("2025-01-01T01:00:00+05:00"), to_timezone="Europe/Berlin")
 84        >>> int(modified_timezone.utcoffset().total_seconds())
 85        3600
 86    """  # noqa: E501

136
137        if not isinstance(object_to_convert, pd.Series):
138            new_object.attrs = object_to_convert.attrs
139            return new_object
140
141        series_object = pd.Series(
142            new_object[object_to_convert.name],
143            index=new_object.index,
144            name=object_to_convert.name,
145        )
146        series_object.attrs = object_to_convert.attrs
